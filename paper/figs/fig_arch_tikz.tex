% Fig. 1 — fail-closed netlist agent (vector, TikZ).
% Requires: \usepackage{tikz} \usetikzlibrary{arrows.meta}
% Layout rules: main flow left-to-right, exceptions downward/upward,
% orthogonal connectors only, no crossings. Units: cm.
\begin{tikzpicture}[
  font=\footnotesize,
  box/.style={draw=black!60, line width=0.5pt, rounded corners=2pt,
              align=center, minimum height=0.9cm, inner sep=2.5pt},
  llm/.style={box, fill=orange!14},
  det/.style={box, fill=blue!7},
  gry/.style={box, fill=gray!12},
  bad/.style={box, fill=red!9},
  flow/.style={-{Stealth[length=4.2pt]}, line width=0.6pt, black!85},
  retry/.style={-{Stealth[length=4.2pt]}, line width=0.6pt, red!70!black, dashed},
  fallback/.style={-{Stealth[length=4.2pt]}, line width=0.6pt, black!50, dashed},
  reject/.style={-{Stealth[length=4.2pt]}, line width=0.6pt, red!70!black},
  read/.style={-{Stealth[length=3.6pt]}, line width=0.45pt, black!40, densely dashed},
  lab/.style={font=\scriptsize, text=black!75},
  rlab/.style={font=\scriptsize, text=red!70!black},
]
\newcommand{\sub}[1]{{\scriptsize #1}}

% ===================== (a) Intent routing =====================
\node[gry, minimum width=3.6cm] (ctx) at (2.3,1.62)
  {\textbf{Prompt context}\\\sub{Full catalog (always)}\\\sub{+ BM25 top-$k$ examples (additive)}};
\node[llm, minimum width=2.8cm] (llm) at (2.3,0)
  {LLM intent router\\\sub{(frozen mini-class model)}};
\node[det, minimum width=2.4cm] (val) at (5.6,0)
  {Validate intent\\\sub{(schema +}\\\sub{live-state grounding)}};
\node[det, minimum width=1.5cm] (disp) at (8.0,0) {Dispatch};
\node[gry, minimum width=2.1cm] (noop) at (5.25,-1.78)
  {Safe no-op\\\sub{(design unchanged)}};

\draw[flow] (-0.25,0) -- (llm.west) node[lab, pos=0.45, above] {NL request};
\draw[flow] (ctx) -- (llm);
\draw[flow] (llm) -- (val);
\draw[flow] (val) -- (disp);
% first failure: one corrective retry back to the router
\draw[retry] (val.south -| 4.6,0) -- (4.6,-0.92) -- (2.3,-0.92) -- (llm.south);
\node[rlab, anchor=north west] at (2.38,-0.95) {invalid $\rightarrow$ retry once};
% still invalid after the retry: terminal safe no-op
\draw[fallback] (val.south -| 5.25,0) -- (noop.north)
  node[lab, pos=0.42, right] {still invalid};

% ===================== (b) Guarded execution =====================
\node[det, minimum width=1.7cm] (ana) at (10.0,-1.5)
  {Analysis\\\sub{(read-only)}};
\node[box, fill=white, minimum width=1.0cm] (reply) at (11.55,-1.5) {Reply};
\node[det, minimum width=1.9cm] (snap) at (12.75,0)
  {Snapshot\\+ transform};
\node[det, minimum width=2.4cm] (guard) at (15.45,0)
  {Guards\\\sub{(CEC $\cdot$ bounds $\cdot$ basis)}};
\node[det, minimum width=1.8cm] (commit) at (15.45,-1.5)
  {Commit\\\sub{(delta ledger)}};
\node[bad, minimum width=2.6cm] (rej) at (15.45,1.55)
  {Reject / discard\\candidate\\\sub{(netlist unchanged)}};

\draw[flow] (disp) -- (snap);
\draw[flow] (disp.south) |- (ana.west);
\draw[flow] (ana) -- (reply);
\draw[flow] (snap) -- (guard);
\draw[flow] (guard) -- (commit) node[lab, midway, right] {PASS};
\draw[reject] (guard) -- (rej) node[rlab, midway, right] {FAIL};

% ===================== persistent netlist state =====================
\node[gry, minimum width=10.35cm, minimum height=0.55cm] (state) at (11.55,-2.72)
  {Persistent netlist state};
\draw[read] (state.north -| 6.55,0) -- (val.south -| 6.55,0) node[lab, pos=0.5, right] {read};
\draw[read] (state.north -| ana) -- (ana.south);
\draw[read] (state.north -| snap) -- (snap.south) node[lab, pos=0.5, right] {read};
\draw[flow] (commit) -- (state.north -| commit) node[lab, midway, right] {write};

% ===================== panel titles / divider =====================
\draw[black!22, line width=0.4pt] (9.0,2.55) -- (9.0,-2.3);
\node[font=\small\bfseries, anchor=west] at (-0.3,2.6) {(a) Intent routing};
\node[font=\small\bfseries, anchor=west] at (9.2,2.6) {(b) Guarded execution};
\end{tikzpicture}
